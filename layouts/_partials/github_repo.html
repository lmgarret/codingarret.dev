(dict "headers" $headers)) }} {{- with .Err }} {{- warnf "projects: could not fetch %s: %s" $ . }} {{- else with .Value }} {{- $data = . | transform.Unmarshal }} {{- end }} {{- end }} {{- return $data -}}
